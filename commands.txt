# ========================================
# Amazon Sales Lakehouse Project
# Complete Launch Commands
# ========================================

# STEP 1: Clean Environment
# Remove all containers, volumes, and orphaned containers
docker compose down -v --remove-orphans

# STEP 2: Build Custom Images
# Build spark-jobs image (Spark batch + streaming jobs)
docker compose --profile build build
# Build producer image (Kafka CSV producer)
docker compose --profile manual build

# STEP 3: Start All Services
# Launch the entire stack (Kafka, HDFS, Spark, Hive, Trino, Postgres, Airflow, Streamlit)
docker compose up -d

# STEP 4: Check Container Status
# Verify all containers are running
docker ps --format "table {{.Names}}\t{{.Status}}\t{{.Ports}}"

# Wait ~90 seconds for all services to initialize, especially:
# - HDFS namenode/datanode + hdfs-init (directory creation)
# - hive-metastore-init (schematool schema creation)
# - Hive metastore + hive-server (table creation)
# - Airflow database initialization

# STEP 5: Restart Airflow Scheduler (if needed)
# The scheduler may exit if it starts before the webserver finishes `airflow db init`.
# Check with: docker ps -a | findstr airflow-scheduler
# If it shows "Exited", restart it:
docker start airflow-scheduler

# ========================================
# Verification Commands
# ========================================

# Kafka: Check messages in topic
docker exec -it kafka kafka-console-consumer --bootstrap-server kafka:9092 --topic amazon-sales --from-beginning --max-messages 5

# HDFS: Check Bronze layer (CSV loaded directly from Amazon.csv)
docker exec -it namenode bash -lc "/opt/hadoop-3.1.3/bin/hdfs dfs -ls /datalake/bronze/amazon_sales_csv | tail"

# HDFS: Check Silver layer (Parquet - cleaned/enriched data)
docker exec -it namenode bash -lc "/opt/hadoop-3.1.3/bin/hdfs dfs -ls /datalake/silver/amazon_sales | tail"

# HDFS: Check Gold layer (Parquet - aggregated KPIs)
docker exec -it namenode bash -lc "/opt/hadoop-3.1.3/bin/hdfs dfs -ls /datalake/gold/amazon_sales_gold | tail"

# Postgres: Check aggregated sales count
docker exec -it postgres-mart psql -U mart -d martdb -c "SELECT count(*) FROM orders_live;"

# Postgres: View sample aggregated data
docker exec -it postgres-mart psql -U mart -d martdb -c "SELECT day, category, country, totalorders, totalrevenue FROM orders_live LIMIT 10;"

# Postgres: Distinct dimensions
docker exec -it postgres-mart psql -U mart -d martdb -c "SELECT count(DISTINCT day) AS days, count(DISTINCT category) AS categories, count(DISTINCT country) AS countries FROM orders_live;"

# Hive: List tables in metastore
docker exec -it hive-server bash -lc "/opt/hive/bin/beeline -u jdbc:hive2://localhost:10000 -n hive -e 'SHOW TABLES IN datalake;'"

# Trino: Query Gold layer
docker exec trino trino --execute "SELECT count(*) FROM hive.datalake.gold_amazon_sales_daily"
docker exec trino trino --execute "SELECT * FROM hive.datalake.gold_amazon_sales_daily LIMIT 5"

# ========================================
# Airflow DAG Execution
# ========================================

# After logging into Airflow (http://localhost:8088):
# 1. Enable the DAG: amazon_dual_pipeline_orchestration
# 2. Trigger the DAG manually
# 3. Monitor task execution (typical full run ~2 minutes)
# 4. Check task logs for any failures

# DAG Task Flow:
#   start
#   |-- produce_csv_to_kafka  (CSV -> Kafka ~22s)
#   |   `-- spark_oltp_streaming  (Kafka -> aggregate -> Postgres ~80s)
#   `-- csv_to_bronze_hdfs  (CSV -> HDFS Bronze ~11s)
#       `-- bronze_to_silver  (Bronze -> Silver Parquet ~26s)
#           `-- silver_to_gold  (Silver -> Gold Parquet ~80s)
#   [spark_oltp_streaming, silver_to_gold] >> end
